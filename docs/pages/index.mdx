## A sortable list of 1,000 rows in one template

Only the rows on screen are in the DOM. Drops arrive as events, and a helper applies them to your signal.

1. **Install the package** No providers and no global stylesheet to register.
2. **Put rows and lists in a group** Declare the row template inside a `vdndGroup` element so every draggable finds its group.
3. **Apply the drop** The library never changes your data. `reorderItems()` applies the move to your signal.

[Read the quick start →](/guide/start/quick-start)

```angular-html title="template.html" {13}
<div vdndGroup="tasks">
  <ng-template #taskTpl let-task>
    <div class="row" [vdndDraggable]="task.id">{{ task.title }}</div>
  </ng-template>

  <vdnd-sortable-list
    droppableId="backlog"
    [items]="tasks()"
    [itemHeight]="48"
    [containerHeight]="336"
    [itemIdFn]="taskId"
    [itemTemplate]="taskTpl"
    (drop)="onDrop($event)"
  />
```

```ts title="quick-start.ts" {9}
export class QuickStartExampleComponent {
  readonly tasks = signal<Task[]>(
    Array.from({ length: 1000 }, (_, i) => ({ id: `task-${i + 1}`, title: `Task ${i + 1}` })),
  );

  readonly taskId = (task: Task): string => task.id;

  onDrop(event: DropEvent): void {
    reorderItems(event, this.tasks);
  }
}
```
